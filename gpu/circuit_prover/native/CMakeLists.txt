cmake_minimum_required(VERSION 3.28)
# Cmake must be at least 3.28 - the earlier one didn't pass cuda_standard properly.
project(circuit_prover_native)
enable_language(CUDA)
if ((NOT DEFINED CMAKE_CUDA_ARCHITECTURES) OR (CMAKE_CUDA_ARCHITECTURES STREQUAL ""))
    set(CMAKE_CUDA_ARCHITECTURES native)
endif ()
option(AB_DETERMINISTIC_POW "Enable deterministic PoW search" OFF)
option(GPU_PROVER_ENABLE_LINEINFO "Enable lineinfo for CUDA profiling builds" OFF)
# Main native archive (NTT now lives in the gpu_ntt crate's own archive).
add_library(circuit_prover_native STATIC)
add_subdirectory(ops)
add_subdirectory(prover)
add_subdirectory(primitives)
add_subdirectory(witness)
# Bench-only GKR eval-ISA kernels. Compiled into circuit_prover_native (not a
# separate archive) so they share its device-link module and its __constant__
# symbols; gated so production builds never include them.
option(AB_GKR_BENCH "Compile GKR eval-ISA bench kernels" OFF)
if (AB_GKR_BENCH)
    add_subdirectory(bench)
endif ()
foreach (circuit_prover_target circuit_prover_native)
    target_include_directories(${circuit_prover_target} PRIVATE ${GPU_CORE_NATIVE_INCLUDE} ${GPU_HASH_NATIVE_INCLUDE})
    set_target_properties(${circuit_prover_target} PROPERTIES CUDA_STANDARD 20)
    set_target_properties(${circuit_prover_target} PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
    set_target_properties(${circuit_prover_target} PROPERTIES CUDA_RESOLVE_DEVICE_SYMBOLS ON)
    if (AB_DETERMINISTIC_POW)
        target_compile_definitions(${circuit_prover_target} PRIVATE AB_DETERMINISTIC_POW=1)
    endif ()
    # CCCL headers in CUDA 12+ require MSVC's standard conforming preprocessor.
    if (MSVC)
        target_compile_options(${circuit_prover_target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/Zc:preprocessor>)
    endif ()
    target_compile_options(${circuit_prover_target} PRIVATE --expt-relaxed-constexpr)
    target_compile_options(${circuit_prover_target} PRIVATE --ptxas-options=-v)
    if (GPU_PROVER_ENABLE_LINEINFO)
        target_compile_options(${circuit_prover_target} PRIVATE -lineinfo)
    endif ()
    target_compile_options(${circuit_prover_target} PRIVATE --keep)
endforeach ()
install(TARGETS circuit_prover_native DESTINATION .)
